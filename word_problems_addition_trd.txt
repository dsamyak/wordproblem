1. TECHNICAL OVERVIEW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

This document specifies the technical architecture, component design, data models,
simulation specifications, and implementation standards for the "Word Problems Using
Addition" interactive lesson module (Lesson 2.5) for Intellia SG's Grade 1 Math program.

The module is a single-file React application (JSX artifact/component), designed to be
embedded into the Intellia SG WordPress platform. It follows the identical visual theme,
section architecture, gamification structure, and UX patterns established at
https://readingnumbers.vercel.app/ (Lesson 1.2 — Reading and Writing Numbers).

4.1 Question Schema
--------------------

typescript
interface WordProblemQuestion {
  id: string;                        // e.g. "WP_Q1_001" (type + index)
  type: WordProblemQuestionType;     // Enum: 10 types (Q1–Q10)
  phase: 'practice' | 'challenge';
  difficulty: 1 | 2 | 3;            // 1 = easy, 2 = medium, 3 = hard
  numberRange: 'within10' | 'within20' | 'within100';

  // Problem content
  questionText: string;              // Full word problem text (≤25 words)
  subjectName?: string;              // Singapore context name (Mia, Raju, etc.)
  setting?: string;                  // Context: playground, classroom, hawker etc.
  emoji?: string;                    // Object emoji for visual reinforcement (🍎, 🐟 etc.)

  // Visual model
  visualType: VisualType;            // 'story_scene' | 'number_bond' | 'bar_model' | 'number_line'
  part1: number;                     // First addend (Part A)
  part2: number;                     // Second addend (Part B)
  whole: number;                     // Sum / answer (Whole)
  equation: string;                  // e.g. "6 + 4 = 10"

  // Answer choices
  options: string[];                 // 4 MCQ options (strings, includes correct)
  correctAnswer: string;             // Must be one of options[]

  // Hints (graduated)
  hint1: string;                     // Keyword highlight instruction
  hint2: string;                     // Bar model scaffold description
  explanation: string;               // Full step-by-step shown after 2 wrong attempts

  // Metadata
  singaporeContext: boolean;         // Whether question uses SG names/settings
  keyWords: string[];                // Addition signal words: "altogether", "total" etc.
}

type WordProblemQuestionType =
  | 'simple_altogether'              // Q1: Two groups → find total within 20
  | 'how_many_in_all'               // Q2: Count two sets → sum within 20
  | 'one_digit_plus_one_digit'      // Q3: Simple 1d+1d in context within 10
  | 'one_digit_plus_two_digit'      // Q4: Mixed digits within 100
  | 'two_digit_plus_one_digit'      // Q5: Reverse mixed digits within 100
  | 'join_receive_more'             // Q6: Someone receives more → within 20
  | 'total_cost'                    // Q7: Adding amounts in cents within 100
  | 'combined_lengths'              // Q8: Adding measurement values within 20
  | 'number_bond_missing_part'      // Q9: Find missing part given whole within 20
  | 'two_sentence_story';           // Q10: Multi-sentence, equation writing, within 100

type VisualType = 'story_scene' | 'number_bond' | 'bar_model' | 'number_line' | 'mixed';

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

5.1 Question Shuffling Algorithm (Fisher-Yates)
------------------------------------------------

javascript
function shuffleQuestions(bank) {
  const shuffled = [...bank];
  for (let i = shuffled.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
  }
  return shuffled;
}

function buildChallengeSet(fullBank) {
  // Ensure minimum 1 question from each type in every session
  const byType = {};
  fullBank.forEach(q => {
    if (!byType[q.type]) byType[q.type] = [];
    byType[q.type].push(q);
  });

  const guaranteed = Object.values(byType).map(arr => {
    const sh = shuffleQuestions(arr);
    return sh[0]; // One from each type guaranteed
  });

  const remaining = fullBank.filter(q => !guaranteed.find(g => g.id === q.id));
  const shuffledRemaining = shuffleQuestions(remaining).slice(0, 90);

  return shuffleQuestions([...guaranteed, ...shuffledRemaining]);
}

5.2 MCQ Option Generation
--------------------------

javascript
function generateOptions(correctAnswer, part1, part2, numberRange) {
  const correct = parseInt(correctAnswer);
  const distractors = new Set();
  
  // Plausible wrong answers: near-miss values
  const candidates = [
    correct - 1,                   // Off-by-one (very common child error)
    correct + 1,                   // Off-by-one
    part1 + part2 - 1,             // Computation error
    Math.max(part1, part2),        // Chose largest part instead of whole
  ].filter(d => d > 0 && d !== correct);

  candidates.forEach(c => {
    if (distractors.size < 3) distractors.add(String(c));
  });

  // Fill remaining slots if needed
  while (distractors.size < 3) {
    const offset = Math.ceil(Math.random() * 3) * (Math.random() > 0.5 ? 1 : -1);
    const d = correct + offset;
    if (d > 0 && d !== correct) distractors.add(String(d));
  }

  const all = shuffleQuestions([correctAnswer, ...[...distractors]]);
  return all.slice(0, 4);
}

const useHintSystem = (question) => {
  const [hintLevel, setHintLevel] = useState(0);
  const [attemptsOnQuestion, setAttemptsOnQuestion] = useState(0);

  const onWrongAnswer = () => {
    const next = attemptsOnQuestion + 1;
    setAttemptsOnQuestion(next);

    if (next === 1) {
      setHintLevel(1);            // Show keyword highlight
      speak(question.hint1);
    } else if (next === 2) {
      setHintLevel(2);            // Show bar model scaffold
      speak(question.hint2);
    } else if (next >= 3) {
      setHintLevel(3);            // Reveal answer + explanation
      speak(question.explanation);
    }
  };

  const resetHints = () => {
    setHintLevel(0);
    setAttemptsOnQuestion(0);
  };

  return { hintLevel, onWrongAnswer, resetHints };
};

10.1 XP Calculation
--------------------

javascript
const XP_VALUES = {
  correctFirstTry: 10,
  correctSecondTry: 7,
  correctWithHint: 5,
  streakBonusThreshold: 5,
  streakBonusAmount: 5,
};

function calculateXP(attemptNumber, hintsUsed, currentStreak) {
  let base;
  if (attemptNumber === 1) base = XP_VALUES.correctFirstTry;
  else if (hintsUsed === 0) base = XP_VALUES.correctSecondTry;
  else base = XP_VALUES.correctWithHint;

  const streakBonus = currentStreak >= XP_VALUES.streakBonusThreshold
    ? XP_VALUES.streakBonusAmount : 0;

  return base + streakBonus;
}

function speak(text, onEnd = null) {
  if (!window.speechSynthesis) return;
  window.speechSynthesis.cancel();                    // Stop any current speech

  const utterance = new SpeechSynthesisUtterance(text);
  utterance.rate = 0.85;                              // Slower — age-appropriate
  utterance.pitch = 1.1;                              // Slightly higher pitch
  utterance.lang = 'en-SG';                           // Singapore English locale

  if (onEnd) utterance.onend = onEnd;

  window.speechSynthesis.speak(utterance);
}

function createTone(frequency, duration = 200, type = 'sine') {
  const ctx = new (window.AudioContext || window.webkitAudioContext)();
  const osc = ctx.createOscillator();
  const gainNode = ctx.createGain();

  osc.type = type;
  osc.frequency.setValueAtTime(frequency, ctx.currentTime);
  gainNode.gain.setValueAtTime(0.3, ctx.currentTime);
  gainNode.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + duration / 1000);

  osc.connect(gainNode);
  gainNode.connect(ctx.destination);
  osc.start();
  osc.stop(ctx.currentTime + duration / 1000);
}

const loadPersistedState = () => {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return null;
    const data = JSON.parse(raw);
    const AGE_LIMIT = 24 * 60 * 60 * 1000; // 24 hours
    if (Date.now() - data.timestamp > AGE_LIMIT) {
      localStorage.removeItem(STORAGE_KEY);
      return null;
    }
    return data;
  } catch (e) {
    return null;
  }
};

const clearProgress = () => {
  localStorage.removeItem(STORAGE_KEY);
};
